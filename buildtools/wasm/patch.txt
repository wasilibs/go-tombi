+// WASI has no sockets, so schema fetching is delegated to the host through the
+// "tombi" import module. The host writes the response body into guest memory it
+// allocates via the exported `tombi_wasm_alloc`, returning the pointer and
+// length through the out-parameters and the HTTP status as the result.
+mod host {
+    #[link(wasm_import_module = "tombi")]
+    unsafe extern "C" {
+        pub fn fetch_schema(
+            url_ptr: *const u8,
+            url_len: usize,
+            out_ptr: *mut usize,
+            out_len: *mut usize,
+        ) -> i32;
+    }
+}

+#[derive(Debug, Clone, Default)]

+
+impl WasiHostHttpClient {
+    pub fn new() -> Self {
+        Self
+    }

+impl HttpClient for WasiHostHttpClient {
+    fn get_bytes<'a>(&'a self, url: &'a str) -> HttpFuture<'a, Result<Bytes, FetchError>> {
+        async move {
+            let mut out_ptr = 0;
+            let mut out_len = 0;
+            // SAFETY: the pointers reference live locals and the URL remains
+            // valid for the duration of the synchronous host call.
+            let status =
+                unsafe { host::fetch_schema(url.as_ptr(), url.len(), &mut out_ptr, &mut out_len) };
+
+            let body = read_host_bytes(out_ptr, out_len);
+
+            if status == 0 {
+                return Err(FetchError::FetchFailed {
+                    reason: "host fetch transport error".to_string(),
+                });
+            }
+            if !(200..300).contains(&status) {
+                return Err(FetchError::StatusNotOk {
+                    status: status as u16,
+                });
+            }
+
+            Ok(Bytes::from(body))
+        }
+        .boxed()
+    }
